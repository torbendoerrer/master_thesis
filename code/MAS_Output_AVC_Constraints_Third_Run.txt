    $self.K40_KO_code = '08':
    ($self.K40_fg_af = '3' OR $self.K40_fg_af = '4') AND
    $self.K40_brm = '0' AND $self.K40_ts != '1',

RESTRICT (V_TW0011)
    $self.K40_KO_code = '09':
    $self.K40_fg_af = '1' AND $self.K40_brm = '0' AND $self.K40_ts != '1' AND
    $self.K40_pl_af = 'd' AND $self.K40_pl IN ('d1','d2','H1','H3') AND $self.K40_abl = '0',

RESTRICT (0000003046)
    $self.K40_KO_code = '99':
    $self.K40_fg_af IN ('1','2','3','4') AND $self.K40_pl IN ('d1','h1','h2') AND
    $self.K40_pl IN ('d2','h3') AND $self.K40_fst = '0'
